// Package controllerdemo provides an independently written example controller
// and a model-based test adapter. These are illustrative APIs, not CAPI APIs.
package controllerdemo

import (
	"context"
	"fmt"

	apiv1 "generative-architecture/examples/machinevendor/api/v1alpha1"

	apierrors "k8s.io/apimachinery/pkg/api/errors"
	metav1 "k8s.io/apimachinery/pkg/apis/meta/v1"
	"k8s.io/apimachinery/pkg/apis/meta/v1/unstructured"
	"k8s.io/apimachinery/pkg/runtime/schema"
	"sigs.k8s.io/controller-runtime/pkg/client"
	"sigs.k8s.io/controller-runtime/pkg/reconcile"
)

var infraGVK = schema.GroupVersionKind{Group: "demo.architecture.local", Version: "v1alpha1", Kind: "InfraMachine"}
var claimGVK = schema.GroupVersionKind{Group: "demo.architecture.local", Version: "v1alpha1", Kind: "MachineClaim"}

type Assignment = apiv1.Assignment

func object(gvk schema.GroupVersionKind, key client.ObjectKey) *unstructured.Unstructured {
	o := &unstructured.Unstructured{Object: map[string]any{}}
	o.SetGroupVersionKind(gvk)
	o.SetNamespace(key.Namespace)
	o.SetName(key.Name)
	return o
}
func get(ctx context.Context, c client.Reader, gvk schema.GroupVersionKind, key client.ObjectKey) (*unstructured.Unstructured, error) {
	o := object(gvk, key)
	return o, c.Get(ctx, key, o)
}
func str(o *unstructured.Unstructured, fields ...string) string {
	v, _, _ := unstructured.NestedString(o.Object, fields...)
	return v
}
func wanted(o *unstructured.Unstructured) bool {
	v, _, _ := unstructured.NestedBool(o.Object, "spec", "allocationRequested")
	return v
}
func assignmentOf(o *unstructured.Unstructured) Assignment {
	epoch, _, _ := unstructured.NestedInt64(o.Object, "status", "epoch")
	return Assignment{MachineID: str(o, "status", "machineId"), Epoch: epoch}
}
func setAssignment(o *unstructured.Unstructured, a Assignment) {
	o.Object["status"] = map[string]any{"machineId": a.MachineID, "epoch": a.Epoch}
}

// MACReconciler owns the MAC behavior. Client and APIReader may differ in a real
// deployment; the demo uses a fake client and explicitly schedules read snapshots.
// AllowStalePublication is a deliberate test mutation, never a generated rule.
// A fresh read followed by a write is not a cross-object atomic transaction.
// +arch:controller=mac
type MACReconciler struct {
	Client                client.Client
	APIReader             client.Reader
	AllowStalePublication bool
}

var _ reconcile.Reconciler = (*MACReconciler)(nil)

// +arch:action=createClaim
func (r *MACReconciler) EnsureClaim(ctx context.Context, key client.ObjectKey) (bool, error) {
	infra, err := get(ctx, r.Client, infraGVK, key)
	if err != nil {
		return false, err
	}
	if !wanted(infra) {
		return false, fmt.Errorf("allocation not requested")
	}
	if _, err = get(ctx, r.Client, claimGVK, key); err == nil {
		return true, nil
	}
	if !apierrors.IsNotFound(err) {
		return false, err
	}
	claim := object(claimGVK, key)
	claim.SetOwnerReferences([]metav1.OwnerReference{{APIVersion: infraGVK.GroupVersion().String(), Kind: infraGVK.Kind, Name: infra.GetName(), UID: infra.GetUID()}})
	claim.Object["spec"] = map[string]any{"infraMachineName": infra.GetName()}
	setAssignment(claim, Assignment{})
	err = r.Client.Create(ctx, claim)
	return err == nil, err
}

// +arch:action=observe
func (r *MACReconciler) Observe(ctx context.Context, key client.ObjectKey) (Assignment, error) {
	claim, err := get(ctx, r.Client, claimGVK, key)
	if err != nil {
		return Assignment{}, err
	}
	return assignmentOf(claim), nil
}

// +arch:action=publish
func (r *MACReconciler) Publish(ctx context.Context, key client.ObjectKey, observed Assignment) (bool, error) {
	claim, err := get(ctx, r.APIReader, claimGVK, key)
	if err != nil {
		return false, err
	}
	if observed.MachineID == "" {
		return false, nil
	}
	infra, err := get(ctx, r.Client, infraGVK, key)
	if err != nil {
		return false, err
	}
	if err = unstructured.SetNestedField(infra.Object, observed.MachineID, "status", "machineId"); err != nil {
		return false, err
	}
	err = r.Client.Status().Update(ctx, infra)
	return err == nil, err
}

// +arch:action=clear
func (r *MACReconciler) Clear(ctx context.Context, key client.ObjectKey) (bool, error) {
	claim, err := get(ctx, r.APIReader, claimGVK, key)
	if err != nil && !apierrors.IsNotFound(err) {
		return false, err
	}
	if err == nil && assignmentOf(claim).MachineID != "" {
		return false, fmt.Errorf("claim still assigned")
	}
	infra, err := get(ctx, r.Client, infraGVK, key)
	if err != nil {
		return false, err
	}
	if err = unstructured.SetNestedField(infra.Object, "", "status", "machineId"); err != nil {
		return false, err
	}
	err = r.Client.Status().Update(ctx, infra)
	return err == nil, err
}

// Reconcile composes the same phase methods exercised by the MBT scheduler.
// MBT drives phases directly to expose the observation/publication interval.
func (r *MACReconciler) Reconcile(ctx context.Context, req reconcile.Request) (reconcile.Result, error) {
	infra, err := get(ctx, r.Client, infraGVK, req.NamespacedName)
	if apierrors.IsNotFound(err) {
		return reconcile.Result{}, nil
	}
	if err != nil {
		return reconcile.Result{}, err
	}
	if !wanted(infra) {
		return reconcile.Result{}, nil
	}
	if _, err = r.EnsureClaim(ctx, req.NamespacedName); err != nil {
		return reconcile.Result{}, err
	}
	observed, err := r.Observe(ctx, req.NamespacedName)
	if err != nil {
		return reconcile.Result{}, err
	}
	if observed.MachineID == "" {
		_, err = r.Clear(ctx, req.NamespacedName)
	} else {
		_, err = r.Publish(ctx, req.NamespacedName, observed)
	}
	return reconcile.Result{}, err
}
